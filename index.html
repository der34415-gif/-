const CONFIG = { SPREADSHEET_NAME: 'سجل تسليم الواجبات', DRIVE_FOLDER_NAME: 'صور واجبات الطلاب' }; function setup() { const props = PropertiesService.getScriptProperties(); const ss = SpreadsheetApp.create(CONFIG.SPREADSHEET_NAME); const folder = DriveApp.createFolder(CONFIG.DRIVE_FOLDER_NAME); props.setProperty('SPREADSHEET_ID', ss.getId()); props.setProperty('FOLDER_ID', folder.getId()); const sheet = ss.getSheets()[0]; sheet.setName('التسليمات'); sheet.appendRow([ 'تاريخ الإرسال', 'اسم الطالب', 'الصف', 'الشعبة', 'عنوان الواجب', 'رابط الصورة', 'اسم الملف' ]); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, 7).setFontWeight('bold'); Logger.log('رابط سجل الواجبات: ' + ss.getUrl()); Logger.log('رابط مجلد الصور: ' + folder.getUrl()); } function doPost(e) { try { const d = JSON.parse(e.postData.contents || '{}'); const name = String(d.studentName || '').trim(); const grade = String(d.grade || '').trim(); const section = String(d.section || '').trim(); const title = String(d.homeworkTitle || '').trim(); const mime = String(d.mimeType || ''); const base64 = String(d.imageBase64 || ''); const filename = String(d.fileName || 'homework.jpg') .replace(/[\\/:*?"<>|]/g, '_'); if (!name || !grade || !section || !base64) { throw new Error('بيانات ناقصة'); } if (!['image/jpeg', 'image/png', 'image/webp'].includes(mime)) { throw new Error('نوع الصورة غير مدعوم'); } if (base64.length > 12 * 1024 * 1024) { throw new Error('الصورة كبيرة جدًا'); } const props = PropertiesService.getScriptProperties(); const folder = DriveApp.getFolderById( props.getProperty('FOLDER_ID') ); const ss = SpreadsheetApp.openById( props.getProperty('SPREADSHEET_ID') ); const blob = Utilities.newBlob( Utilities.base64Decode(base64), mime, filename ); const file = folder.createFile(blob); file.setDescription( 'واجب الطالب: ' + name + ' - ' + grade + ' - ' + section ); ss.getSheetByName('التسليمات').appendRow([ new Date(), name, grade, section, title, file.getUrl(), filename ]); return ContentService.createTextOutput( JSON.stringify({ok: true}) ).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput( JSON.stringify({ok: false, error: String(error)}) ).setMimeType(ContentService.MimeType.JSON); } } function doGet() { return ContentService.createTextOutput( 'بوابة تسليم الواجبات تعمل.' ); }
