const FOLDER_NAME = "صور الواجبات"; const SHEET_NAME = "سجل تسليم الواجبات"; function setup() { const props = PropertiesService.getScriptProperties(); // لا تنشئ مجلدًا وجدولًا جديدين إذا سبق إعداد المشروع. if (props.getProperty("FOLDER_ID") && props.getProperty("SHEET_ID")) { return "تم إعداد المشروع مسبقًا."; } const folder = DriveApp.createFolder(FOLDER_NAME); const spreadsheet = SpreadsheetApp.create(SHEET_NAME); const sheet = spreadsheet.getActiveSheet(); sheet.setName("التسليمات"); sheet.appendRow([ "تاريخ التسليم", "اسم الطالب", "الصف", "الشعبة", "عنوان الواجب", "رابط الصورة" ]); props.setProperties({ FOLDER_ID: folder.getId(), SHEET_ID: spreadsheet.getId() }); return "تم إنشاء مجلد الصور وسجل التسليمات."; } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const props = PropertiesService.getScriptProperties(); const folderId = props.getProperty("FOLDER_ID"); const sheetId = props.getProperty("SHEET_ID"); if (!folderId || !sheetId) { throw new Error("يجب تشغيل setup أولًا."); } if (!data.studentName || !data.grade || !data.section || !data.imageBase64) { throw new Error("بيانات الطالب أو صورة الواجب ناقصة."); } const allowedTypes = [ "image/jpeg", "image/png", "image/webp" ]; if (!allowedTypes.includes(data.mimeType)) { throw new Error("صيغة الصورة غير مدعومة."); } const bytes = Utilities.base64Decode(data.imageBase64); if (bytes.length > 8 * 1024 * 1024) { throw new Error("حجم الصورة يتجاوز 8 ميغابايت."); } const fileName = String(data.fileName || "واجب") .replace(/[\\/:*?"<>|]/g, "_") .substring(0, 150); const blob = Utilities.newBlob( bytes, data.mimeType, fileName ); const folder = DriveApp.getFolderById(folderId); const file = folder.createFile(blob); const spreadsheet = SpreadsheetApp.openById(sheetId); const sheet = spreadsheet.getSheetByName("التسليمات"); sheet.appendRow([ new Date(), String(data.studentName).substring(0, 100), String(data.grade).substring(0, 50), String(data.section).substring(0, 40), String(data.homeworkTitle || "").substring(0, 120), file.getUrl() ]); return jsonResponse({ success: true, message: "تم حفظ الواجب بنجاح." }); } catch (error) { return jsonResponse({ success: false, message: error.message }); } } function doGet() { return jsonResponse({ success: true, message: "خدمة استقبال الواجبات تعمل." }); } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
